// Lesson 021 — Databases, Collections and Documents
// Self-contained mongosh Demo

// STEP 1 — Check MongoDB Connection
db.runCommand({ ping: 1 })

// STEP 2 — Select the Lesson Database
use Lesson021_DB

// STEP 3 — Start Clean
db.dropDatabase()

// STEP 4 — Create Collections
db.createCollection("customers")
db.createCollection("products")
db.createCollection("orders")

// STEP 5 — Display Collections
show collections

// STEP 6 — Create a Customer Document
db.customers.insertOne({
    customerId: "C001",
    name: "John Smith",
    email: "john@example.com",
    status: "ACTIVE"
})

// STEP 7 — View the Customer Document
db.customers.find().pretty()

// STEP 8 — Create a Product Document
db.products.insertOne({
    productId: "P001",
    name: "MongoDB Laptop",
    price: 1200,
    category: "Computers"
})

// STEP 9 — View the Product
db.products.find().pretty()

// STEP 10 — Create a Document with an Embedded Object
db.customers.insertOne({
    customerId: "C002",
    name: "Sarah Wilson",
    email: "sarah@example.com",
    address: {
        street: "100 Main Street",
        city: "Chennai",
        state: "Tamil Nadu",
        country: "India",
        postalCode: "600001"
    }
})

// STEP 11 — View the Embedded Document
db.customers.find({ customerId: "C002" }).pretty()

// STEP 12 — Create a Document with an Array
db.products.insertOne({
    productId: "P002",
    name: "MongoDB Developer Kit",
    price: 850,
    category: "Developer Tools",
    tags: ["mongodb", "developer", "database"]
})

// STEP 13 — View the Array
db.products.find({ productId: "P002" }).pretty()

// STEP 14 — Create a Document with Nested Objects and Arrays
db.customers.insertOne({
    customerId: "C003",
    name: "David Kumar",
    email: "david@example.com",
    address: {
        street: "200 Lake Road",
        city: "Bengaluru",
        state: "Karnataka",
        country: "India"
    },
    preferences: {
        language: "English",
        notifications: true
    },
    tags: ["developer", "mongodb", "java"]
})

// STEP 15 — Display the Complete Customer
db.customers.find({ customerId: "C003" }).pretty()

// STEP 16 — Create an Order Using a Reference
db.orders.insertOne({
    orderId: "O001",
    customerId: "C001",
    productId: "P001",
    quantity: 2,
    totalAmount: 2400,
    status: "CONFIRMED"
})

// STEP 17 — View the Referenced Order
db.orders.find().pretty()

// STEP 18 — Show the Customer Separately
db.customers.find({ customerId: "C001" }).pretty()

// STEP 19 — Show the Product Separately
db.products.find({ productId: "P001" }).pretty()

// STEP 20 — Demonstrate Different Document Shapes
db.customers.find({ customerId: "C001" }).pretty()
db.customers.find({ customerId: "C002" }).pretty()
db.customers.find({ customerId: "C003" }).pretty()

// STEP 21 — Query an Embedded Field
db.customers.find({ "address.city": "Chennai" }).pretty()

// STEP 22 — Query an Array
db.products.find({ tags: "mongodb" }).pretty()

// STEP 23 — Insert Multiple Documents
db.products.insertMany([
    {
        productId: "P003",
        name: "MongoDB Book",
        price: 60,
        category: "Books"
    },
    {
        productId: "P004",
        name: "MongoDB Training",
        price: 300,
        category: "Training"
    }
])

// STEP 24 — Count Documents
db.customers.countDocuments()
db.products.countDocuments()
db.orders.countDocuments()

// STEP 25 — Display Collections
show collections

// STEP 26 — Display All Customers
db.customers.find().pretty()

// STEP 27 — Display All Products
db.products.find().pretty()

// STEP 28 — Display All Orders
db.orders.find().pretty()

// STEP 29 — Demonstrate the MongoDB Hierarchy
db.getName()
show collections
db.customers.findOne()
db.products.findOne()
db.orders.findOne()

// STEP 30 — Final Verification
db.runCommand({
    listCollections: 1
}).cursor.firstBatch.forEach(function(collection) {
    print(collection.name)
})

print("             Embedded Objects")
print("             Arrays")
print("             References")
print("             Flexible Document Structures")
print("==============================================")
