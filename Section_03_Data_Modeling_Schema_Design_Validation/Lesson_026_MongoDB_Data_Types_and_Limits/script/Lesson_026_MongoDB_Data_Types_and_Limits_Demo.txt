use orderdb
db.orders.deleteMany({ orderId: "ORD-1001" })
db.limitDemo.deleteMany({})

// 1. Insert one Order Management document using representative BSON types
db.orders.insertOne({
  orderId: "ORD-1001",
  orderDate: ISODate("2026-09-15T10:30:00Z"),
  status: "CONFIRMED",
  customerId: ObjectId(),
  totalAmount: Decimal128("1001.00"),
  items: [
    { productId: "P100", quantity: NumberInt(2), unitPrice: Decimal128("499.50") },
    { productId: "P200", quantity: NumberInt(1), unitPrice: Decimal128("25.50") }
  ],
  shippingAddress: { city: "Chennai", country: "India" },
  isGift: false,
  notes: null
})

// 2. Read the document
db.orders.findOne({ orderId: "ORD-1001" })

// 3. Inspect BSON types of representative fields
db.orders.aggregate([
  { $match: { orderId: "ORD-1001" } },
  { $project: {
      _id: 0,
      orderIdType: { $type: "$orderId" },
      orderDateType: { $type: "$orderDate" },
      customerIdType: { $type: "$customerId" },
      totalAmountType: { $type: "$totalAmount" },
      itemsType: { $type: "$items" },
      isGiftType: { $type: "$isGift" },
      notesType: { $type: "$notes" }
  } }
])

// 4. Inspect numeric types inside the items array
db.orders.aggregate([
  { $match: { orderId: "ORD-1001" } },
  { $project: {
      _id: 0,
      firstQuantityType: { $type: { $arrayElemAt: ["$items.quantity", 0] } },
      firstUnitPriceType: { $type: { $arrayElemAt: ["$items.unitPrice", 0] } }
  } }
])

// 5. Show a focused business view
db.orders.findOne(
  { orderId: "ORD-1001" },
  { _id: 0, orderId: 1, orderDate: 1, customerId: 1, totalAmount: 1, items: 1, isGift: 1, notes: 1 }
)

// 6. Demonstrate the 16 MiB BSON document-size boundary
const largeString = "X".repeat(16 * 1024 * 1024)
try {
  db.limitDemo.insertOne({ name: "TooLarge", payload: largeString })
} catch (e) {
  print("Expected document-size error:")
  print(e)
}

// 7. A document below the limit is accepted
const safeString = "X".repeat(15 * 1024 * 1024)
db.limitDemo.insertOne({ name: "WithinLimit", payload: safeString })

// 8. Verify the accepted document
const acceptedDoc = db.limitDemo.findOne({ name: "WithinLimit" })
print({ accepted: acceptedDoc !== null, approximateBsonBytes: Object.bsonsize(acceptedDoc) })

// Cleanup (optional)
db.orders.deleteOne({ orderId: "ORD-1001" })
db.limitDemo.deleteMany({})
