// Lesson 022 — Build with JSON Data
// MongoDB mongosh Demo — JSON Data Only
// Run step by step in mongosh

db.runCommand({ ping: 1 })

use Lesson022_JSON

db.dropDatabase()

// 1. Create a simple JSON-style document
db.customers.insertOne({
  name: "Arun",
  email: "arun@example.com",
  age: 30
})

// 2. View the document
db.customers.find()

// 3. Create a document with nested JSON
db.customers.insertOne({
  name: "Priya",
  email: "priya@example.com",
  address: {
    street: "MG Road",
    city: "Chennai",
    country: "India"
  }
})

// 4. Create a document with a JSON array
db.customers.insertOne({
  name: "Kumar",
  email: "kumar@example.com",
  skills: [
    "MongoDB",
    "Java",
    "Spring Boot"
  ]
})

// 5. Create a document with nested objects and arrays
db.customers.insertOne({
  name: "Meena",
  contact: {
    email: "meena@example.com",
    phone: "9876543210"
  },
  skills: [
    {
      name: "MongoDB",
      level: "Advanced"
    },
    {
      name: "Java",
      level: "Intermediate"
    }
  ]
})

// 6. Insert multiple JSON-style documents
db.products.insertMany([
  {
    name: "Laptop",
    price: 75000,
    category: "Electronics"
  },
  {
    name: "Monitor",
    price: 18000,
    category: "Electronics"
  },
  {
    name: "Keyboard",
    price: 2500,
    category: "Accessories"
  }
])

// 7. Read all documents
db.products.find()

// 8. Read selected JSON fields
db.products.find(
  {},
  { name: 1, price: 1, _id: 0 }
)

// 9. Query a nested JSON field
db.customers.find({
  "address.city": "Chennai"
})

// 10. Query an array value
db.customers.find({
  skills: "MongoDB"
})

// 11. Query an object inside an array
db.customers.find({
  "skills.name": "MongoDB"
})

// 12. Count the JSON documents
db.customers.countDocuments()
db.products.countDocuments()

// 13. View the complete collection structure
show collections

// 14. Final verification
db.customers.find().pretty()
db.products.find().pretty()

// End of Lesson 022 demo
