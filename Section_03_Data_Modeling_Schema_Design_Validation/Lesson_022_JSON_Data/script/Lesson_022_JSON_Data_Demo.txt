// MongoDB mongosh Demo — JSON Data
// Run step by step in mongosh

db.runCommand({ ping: 1 })

use JSON_Data

db.dropDatabase()

// 1. Create a simple JSON-style document
db.customers.insertOne({
  name: "Avi",
  email: "avi@example.com",
  age: 40,
  location: "Bangalore"
})

// 2. View the document
db.customers.find()

// 3. Create a document with a nested JSON object
db.customers.insertOne({
  name: "Avi",
  email: "avi@example.com",
  address: {
    street: "MG Road",
    city: "Bangalore",
    state: "Karnataka",
    country: "India"
  }
})

// 4. Create a document with a JSON array
db.customers.insertOne({
  name: "Avi",
  location: "Bangalore",
  skills: [
    "MongoDB",
    "Java",
    "Spring Boot"
  ]
})

// 5. Create a document with nested objects and arrays
db.customerProfiles.insertOne({
  name: "Avi",
  location: {
    city: "Bangalore",
    state: "Karnataka",
    country: "India"
  },
  contact: {
    email: "avi@example.com",
    phone: "9876543210"
  },
  skills: [
    {
      name: "MongoDB",
      level: "Advanced"
    },
    {
      name: "Java",
      level: "Intermediate"
    }
  ]
})

// 6. Create a JSON-style Order Management document
db.orders.insertOne({
  orderId: "ORD-1001",
  customer: {
    name: "Avi",
    city: "Bangalore"
  },
  status: "CONFIRMED",
  items: [
    {
      sku: "LAP-100",
      name: "Laptop",
      qty: 1,
      price: 75000
    },
    {
      sku: "MOU-200",
      name: "Mouse",
      qty: 2,
      price: 1500
    }
  ]
})

// 7. View the order document
db.orders.find()

// 8. Insert multiple JSON-style product documents
db.products.insertMany([
  {
    name: "Laptop",
    price: 75000,
    category: "Electronics"
  },
  {
    name: "Monitor",
    price: 18000,
    category: "Electronics"
  },
  {
    name: "Keyboard",
    price: 2500,
    category: "Accessories"
  }
])

// 9. Read all products
db.products.find()

// 10. Read selected JSON fields
db.products.find(
  {},
  {
    name: 1,
    price: 1,
    _id: 0
  }
)

// 11. Query a nested JSON field
db.customers.find({
  "address.city": "Bangalore"
})

// 12. Query Avi by nested location
db.orders.find({
  "customer.city": "Bangalore"
})

// 13. Query an array value
db.customers.find({
  skills: "MongoDB"
})

// 14. Query an object inside an array
db.customerProfiles.find({
  "skills.name": "MongoDB"
})

// 15. Query a nested order item
db.orders.find({
  "items.sku": "MOU-200"
})

// 16. Read only selected fields from the order
db.orders.find(
  { orderId: "ORD-1001" },
  {
    _id: 0,
    orderId: 1,
    status: 1,
    "customer.name": 1,
    "customer.city": 1
  }
)

// 17. Count the JSON documents
db.customers.countDocuments()
db.customerProfiles.countDocuments()
db.orders.countDocuments()
db.products.countDocuments()

// 18. View the collections
show collections

// 19. Final verification
db.customers.find().pretty()
db.customerProfiles.find().pretty()
db.orders.find().pretty()
db.products.find().pretty()

// End of demo
